function sortFolders(folders: Folder[]) {
	return [...folders].sort((a, b) => {
		if (a.starred !== b.starred) return a.starred ? -1 : 1;
		return b.createdAt - a.createdAt;
	});
}
function sortRoles(roles: Role[]) {
	return [...roles].sort((a, b) => {
		if (a.starred !== b.starred) return a.starred ? -1 : 1;
		return b.updatedAt - a.updatedAt;
	});
}

function cloneMessage(m: ChatMessage) {
	return {
		...m,
		id: uid("msg"),
		image: m.image ? { ...m.image } : undefined,
		images: m.images?.map((im) => ({ ...im })),
		replyVersions: m.replyVersions?.map((v) => ({
			...v,
			id: uid("ver"),
			bubbles: v.bubbles.map((b) => ({ ...b })),
			images: v.images?.map((im) => ({ ...im })),
		})),
	};
}
function hydrateMessage(m: ChatMessage) {
	const images = m.images?.length ? m.images : m.image ? [m.image] : undefined;
	const imageIndex = images ? Math.min(Math.max(m.imageIndex ?? images.length - 1, 0), images.length - 1) : undefined;
	return {
		...m,
		images,
		image: m.image ?? (images && imageIndex != null ? images[imageIndex] : undefined),
		imageIndex,
		imagePending: false,
		imagePhase: undefined,
		imageBusy: undefined
	};
}
function hydrateRoles(roles: Role[] | undefined) {
	return (roles ?? []).map((r) => ({
		...r,
		draft: normalizeDraft(r.draft),
		imageParams: cloneImageParams(r.imageParams),
		messages: (r.messages ?? []).map(hydrateMessage),
		sort: typeof r.sort === "number" ? r.sort : 0
	}));
}
let persistReady = false;
export function markPersistReady() {
	persistReady = true;
}
function countPersistedRoles(raw: string | null) {
	if (!raw) return 0;
	try {
		const parsed = JSON.parse(raw);
		return Array.isArray(parsed?.state?.roles) ? parsed.state.roles.length : 0;
	} catch {
		return 0;
	}
}
function parsePersist(raw: string | null): { state?: any } | null {
	if (!raw) return null;
	try {
		const v = JSON.parse(raw);
		return v && typeof v === "object" ? v : null;
	} catch {
		return null;
	}
}
function persistScore(p: { state?: { savedAt?: number; roles?: unknown } } | null) {
	if (!p?.state) return -1;
	const savedAt = Number(p.state.savedAt) || 0;
	const roles = Array.isArray(p.state.roles) ? p.state.roles.length : 0;
	return savedAt > 0 ? savedAt : roles;
}
function mergePersistBlobs(idb: string | null, ls: string | null) {
	const a = parsePersist(idb);
	const b = parsePersist(ls);
	if (!a && !b) return null;
	if (!a) return ls;
	if (!b) return idb;
	const primary = persistScore(a) >= persistScore(b) ? a : b;
	const other = primary === a ? b : a;
	if (primary.state) {
		primary.state.appearanceCards = mergeCardLists(primary.state.appearanceCards, other.state?.appearanceCards);
		primary.state.personaCards = mergeCardLists(primary.state.personaCards, other.state?.personaCards);
	}
	try {
		return JSON.stringify(primary);
	} catch {
		return persistScore(a) >= persistScore(b) ? idb : ls;
	}
}
const dualStorage: StateStorage = {
	getItem: async (name: string) => {
		let idb = null;
		let ls = null;
		try {
			idb = await Promise.race([
				kvGet(name),
				new Promise<string | null>((resolve) => {
					window.setTimeout(() => resolve(null), 8000);
				}),
			]);
		} catch {
			idb = null;
		}
		try {
			ls = localStorage.getItem(name);
		} catch {
			ls = null;
		}
		const merged = mergePersistBlobs(idb, ls);
		if (merged) return merged;
		const nI = countPersistedRoles(idb);
		const nL = countPersistedRoles(ls);
		if (nI >= nL && idb) return idb;
		return ls ?? idb;
	},
	setItem: async (name: string, value: string) => {
		if (!persistReady) return;
		try {
			await kvSet(name, value);
		} catch {
			/* */
		}
		try {
			localStorage.setItem(name, value);
		} catch {
			/* */
		}
		try {
			const parsed = parsePersist(value);
			const appear = parsed?.state?.appearanceCards;
			const persona = parsed?.state?.personaCards;
			if (Array.isArray(appear)) void writeAppearanceCards(appear);
			if (Array.isArray(persona)) void writePersonaCards(persona);
		} catch {
			/* */
		}
	},
	removeItem: async (name: string) => {
		if (!persistReady) return;
		try {
			await kvDel(name);
		} catch {
			/* */
		}
		try {
			localStorage.removeItem(name);
		} catch {
			/* */
		}
	}
};

export function paramsRoleId(s: {
	composer: AppState["composer"];
	editingRoleId: string | null;
	activeRoleId: string | null;
	draftRoleId: string | null;
	roles: { id: string }[];
}): string | null {
	if (s.composer === "edit" && s.editingRoleId) return s.editingRoleId;
	if (s.composer === "new" || s.composer === "assemble") return s.activeRoleId || s.draftRoleId;
	return s.activeRoleId ?? s.roles[0]?.id ?? null;
}

			updatedAt: Date.now()

			if (setIds.has(f.id) && f.starred) newlyStarred.push({ id: f.id, kind: "f", parentId: folderParent(f) });
		}
		for (const r of roles) {
			if (setIds.has(r.id) && r.starred) newlyStarred.push({ id: r.id, kind: "r", parentId: r.folderId ?? null });
		}

			const k = s.parentId ?? "";
			const arr = byParent.get(k) ?? [];
			arr.push(s);
			byParent.set(k, arr);
		}

			const parentId = k === "" ? null : k;
			const fs = group.filter((s) => s.kind === "f");
			const rs = group.filter((s) => s.kind === "r");
			let topF = nextSortTop(folders, roles, parentId, "folder");
			for (const s of fs) {
				folders = folders.map((f) => f.id === s.id ? { ...f, sort: topF } : f);
				topF -= 10;
			}
			let topR = nextSortTop(folders, roles, parentId, "role");
			for (const s of rs) {
				roles = roles.map((r) => r.id === s.id ? { ...r, sort: topR } : r);
				topR -= 10;
			}
		}
		const nowStarred = newlyStarred.length > 0;
		set({
			folders,
			roles
		});
		showToast(nowStarred ? "已星标并置顶" : "已取消星标");
		get().exitSelect();

			if (!setIds.has(f.id) || blocked(f)) return f;

			fTop += 10;
			return { ...f, parentId: folderId, sort };
		});
		const roles2 = get().roles.map((r) => {
			if (!setIds.has(r.id)) return r;
			const sort = rTop;
			rTop += 10;
			return { ...r, folderId, sort, updatedAt: Date.now() };

			expanded.add(folderId);

				expanded.add(pid);
				seen.add(pid);
				pid = folders.find((f) => f.id === pid)?.parentId ?? null;
			}
		}

		showToast("已移动");
		get().exitSelect();

export async function overlayLibraryCards() {
	const [appearDisk, personaDisk] = await Promise.all([readAppearanceCards(), readPersonaCards()]);
	const s = useAppStore.getState();
	const appearanceCards = appearDisk.length ? mergeCardLists(appearDisk, s.appearanceCards) : s.appearanceCards;
	const personaCards = personaDisk.length ? mergeCardLists(personaDisk, s.personaCards) : s.personaCards;
	if (!appearDisk.length && appearanceCards.length) void writeAppearanceCards(appearanceCards);
	if (!personaDisk.length && personaCards.length) void writePersonaCards(personaCards);
	if (appearanceCards !== s.appearanceCards || personaCards !== s.personaCards) {
		useAppStore.setState({
			appearanceCards,
			personaCards
		});
	}
}
