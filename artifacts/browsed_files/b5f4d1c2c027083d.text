import type { Connection, ImageParams } from "./types";
import { startNaiCooldown } from "./nai-cooldown";

const BACKUP_KEY = "huiyu-nai-conn";
const STORE_KEY = "huiyu-store-v1";

export function defaultConn(): Connection {
  return { naiKey: "", naiBase: "https://api.idlecloud.cc" };
}

export function normalizeBase(url: string): string {
  return (url || "https://api.idlecloud.cc").trim().replace(/\/+$/, "") || "https://api.idlecloud.cc";
}

function cleanConn(c: Partial<Connection> | null | undefined): Connection {
  return {
    naiKey: String(c?.naiKey ?? "").trim(),
    naiBase: normalizeBase(String(c?.naiBase ?? "")),
  };
}

export function readNaiBackup(): Connection {
  if (typeof window === "undefined") return defaultConn();
  try {
    const raw = localStorage.getItem(BACKUP_KEY);
    if (!raw) return defaultConn();
    return cleanConn(JSON.parse(raw) as Partial<Connection>);
  } catch {
    return defaultConn();
  }
}

export function readPersistStoreConn(): Connection {
  if (typeof window === "undefined") return defaultConn();
  try {
    const raw = localStorage.getItem(STORE_KEY);
    if (!raw) return defaultConn();
    const parsed = JSON.parse(raw) as {
      state?: { settings?: { connection?: Partial<Connection> } };
    };
    return cleanConn(parsed?.state?.settings?.connection);
  } catch {
    return defaultConn();
  }
}

export function writeNaiBackup(c: Connection) {
  if (typeof window === "undefined") return;
  try {
    localStorage.setItem(BACKUP_KEY, JSON.stringify(cleanConn(c)));
  } catch {
    /* quota */
  }
}

export function pickConn(...list: Array<Partial<Connection> | null | undefined>): Connection {
  for (const item of list) {
    const c = cleanConn(item);
    if (c.naiKey) return c;
  }
  return cleanConn(list.find(Boolean) ?? defaultConn());
}

export function resolvedConn(fromStore?: Connection | null): Connection {
  return pickConn(fromStore, readNaiBackup(), readPersistStoreConn());
}

export function requestConnectDialog() {
  if (typeof window === "undefined") return;
  window.dispatchEvent(new Event("huiyu:connect"));
}

function sleep(ms: number, signal?: AbortSignal) {
  return new Promise<void>((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException("Aborted", "AbortError"));
      return;
    }
    const t = setTimeout(resolve, ms);
    const onAbort = () => {
      clearTimeout(t);
      reject(new DOMException("Aborted", "AbortError"));
    };
    signal?.addEventListener("abort", onAbort, { once: true });
  });
}

function isAbort(e: unknown, signal?: AbortSignal) {
  return (
    Boolean(signal?.aborted) ||
    (e instanceof DOMException && e.name === "AbortError") ||
    (e instanceof Error && e.name === "AbortError")
  );
}

type NaiApiJson = {
  image?: string;
  seed?: number;
  job_id?: string;
  status_url?: string;
  status?: string;
  failed?: boolean;
  error?: string;
  file_url?: string;
  v5_delivery?: {
    status_url?: string;
    file_url?: string;
    confirm_url?: string;
    confirmation_required?: boolean;
    sha256?: string;
    size_bytes?: number;
  };
};

async function callNaiApi(body: Record<string, unknown>, signal?: AbortSignal): Promise<NaiApiJson> {
  const timeout = AbortSignal.timeout(45_000);
  const combined =
    signal && typeof AbortSignal.any === "function" ? AbortSignal.any([timeout, signal]) : timeout;
  const res = await fetch("/api/nai", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
    signal: combined,
  });
  const raw = await res.text();
  let json: NaiApiJson = {};
  try {
    json = JSON.parse(raw) as NaiApiJson;
  } catch {
    throw new Error(res.ok ? "生图服务返回了异常页面，请稍后重试" : `生图失败（${res.status}）`);
  }
  if (!res.ok || json.error) throw new Error(json.error || "生图失败");
  return json;
}

export async function postNai(opts: {
  params: ImageParams;
  positive: string;
  seed: number;
  key?: string;
  base?: string;
  wait?: boolean;
  signal?: AbortSignal;
  onPhase?: (phase: "submit" | "draw" | "fetch") => void;
}): Promise<{ image: string; seed: number }> {
  const conn = pickConn({ naiKey: opts.key ?? "", naiBase: opts.base ?? "" }, resolvedConn());
  if (!conn.naiKey) {
    requestConnectDialog();
    throw new Error("尚未填写 NAI 中转站 API Key，请点右上角连接并保存");
  }
  if (!opts.positive.trim()) throw new Error("请先填写正面提示词");

  const onPhase = (phase: "submit" | "draw" | "fetch") => {
    if (phase === "fetch") startNaiCooldown();
    opts.onPhase?.(phase);
  };

  try {
    onPhase("submit");
    const submit = await callNaiApi(
      {
        phase: "submit",
        key: conn.naiKey,
        base: conn.naiBase,
        params: opts.params,
        positive: opts.positive,
        seed: opts.seed,
      },
      opts.signal,
    );
    if (submit.image) {
      onPhase("fetch");
      return { image: submit.image, seed: submit.seed ?? opts.seed };
    }
    if (!submit.job_id) throw new Error("中转站没有返回任务");

    onPhase("draw");
    let statusUrl = submit.status_url;
    for (let i = 0; i < 90; i++) {
      if (opts.signal?.aborted) throw new DOMException("Aborted", "AbortError");
      await sleep(2000, opts.signal);
      const poll = await callNaiApi(
        {
          phase: "poll",
          key: conn.naiKey,
          base: conn.naiBase,
          job_id: submit.job_id,
          status_url: statusUrl,
          seed: opts.seed,
        },
        opts.signal,
      );
      if (poll.status_url) statusUrl = poll.status_url;
      if (poll.image) {
        onPhase("fetch");
        return { image: poll.image, seed: poll.seed ?? opts.seed };
      }
      if (poll.failed) throw new Error(poll.error || "生成失败");
      if (poll.status === "ready" && poll.file_url) {
        onPhase("fetch");
        const fetched = await callNaiApi(
          {
            phase: "fetch",
            key: conn.naiKey,
            base: conn.naiBase,
            job_id: submit.job_id,
            file_url: poll.file_url,
            v5_delivery: poll.v5_delivery,
            seed: opts.seed,
          },
          opts.signal,
        );
        if (fetched.image) return { image: fetched.image, seed: fetched.seed ?? opts.seed };
        if (fetched.failed) throw new Error(fetched.error || "回传失败");
      }
    }
    throw new Error("生图超时，请稍后重试");
  } catch (e) {
    if (isAbort(e, opts.signal)) {
      throw e instanceof Error ? e : new DOMException("Aborted", "AbortError");
    }
    throw e instanceof Error ? e : new Error("生图失败");
  }
}
