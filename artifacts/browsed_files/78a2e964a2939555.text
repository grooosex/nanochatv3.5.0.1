function castBlock(draft: RoleDraft): string {
  if (!isMultiDraft(draft) || !draft.cast?.length) {
    return `【角色名】${draft.name || "角色"}
【要求总览】（最高优先级）
${draft.overview || "（无）"}
【人设】
${draft.persona || "（无）"}
【说话方式】
${draft.speech || "（无）"}
【外貌备忘】（只作你脑中的视觉参考，不要在回复里复读发色瞳色；生图提示词也不要写外貌）
${draft.appearance || "（无）"}
【开场场景】
${draft.scene || "（无）"}`;
  }
  const people = draft.cast
    .map((c, i) => {
      return `【角色${i + 1}｜${c.name.trim() || "未命名"}】
人设：
${c.persona || "（无）"}
说话方式：
${c.speech || "（无）"}
外貌备忘（视觉参考，回复和生图都不要复读发色瞳色体型）：
${c.appearance || "（无）"}`;
    })
    .join("\n\n");
  return `【多人】共 ${draft.cast.length} 人，你要同时扮演在场的所有角色。用户是「你」。
【要求总览】（最高优先级）
${draft.overview || "（无）"}
${people}
【开场场景】
${draft.scene || "（无）"}`;
}

export function buildChatSystem(draft: RoleDraft, directorNotes: string[], memoryText?: string): string {
  const extras = draft.extras
    .filter((e) => e.content.trim())
    .map((e, i) => `【${e.title || "新增" + (i + 1)}】\n${e.content.trim()}`)
    .join("\n\n");
  const notes =
    directorNotes.length > 0
      ? `\n\n【导演指令 / 对 Grok 的要求，优先遵守】\n${directorNotes.map((n) => "- " + n).join("\n")}`
      : "";
  const multi = isMultiDraft(draft);
  const group = isGroupMode(draft);
  const names = statusNamesOf(draft);
  const present = (draft.cast ?? []).filter((c) => !c.away).map((c) => c.name.trim() || "角色");
  const away = (draft.cast ?? []).filter((c) => c.away).map((c) => c.name.trim()).filter(Boolean);
  const presence = multi
    ? `\n【在场】${present.join("、") || "（全员）"}${away.length ? `\n【离场，本轮不要让他们说话也不要画他们】${away.join("、")}` : ""}`
    : "";

  const memory =
    memoryText && memoryText.trim()
      ? `\n【长期记忆·更早剧情的摘要，视为已发生的事实，不要遗忘】
${memoryText.trim()}
（后台摘要，不要在回复里提起「记忆」或复读它。后面的对话才是最近原文。）`
      : "";

  return `你是「绘语」里的角色扮演引擎。你同时做两件事：1) 以角色身份推进剧情；2) 为当前画面写 NovelAI 4.5/5 的 Danbooru 风格提示词。

${HARD_RULES}
${group ? "\n" + GROUP_RULES : ""}

${castBlock(draft)}
${presence}
${extras ? extras : ""}
${memory}
${draft.statusOn ? "\n【状态栏开启】每轮回复都要更新状态栏。" + (group ? "状态栏放在旁白末尾。\n" : "\n") + statusTemplateNote(names) + "\n当前状态栏：\n" + (draft.statusText || "") : "【状态栏关闭】不要输出状态栏。"}
${notes}

${multi ? MULTI_PROMPT_RULES : SINGLE_PROMPT_RULES}
${draft.adultPrompt ? "\n" + ADULT_PROMPT_STYLE : ""}`;
}

export function buildFieldRewriteSystem(field: string, multi = false, who = ""): string {

    name: "名字",

    persona: "人设",
    speech: "说话方式",
    appearance: "外貌备忘",
    scene: "开场场景",
  };
  const label = labels[field] || field;
  const appearanceNote =
    field === "appearance"
      ? "外貌备忘必须写成 NovelAI/Danbooru 英文 tag，以 girl 或 boy 开头（不要 1girl），逗号分隔，只写固定外观，不要动作、质量词和画师。\n"

      ? `写这一栏时必须同时做到：

      : "";

}

export function openingUserLine(draft: RoleDraft): string {
  const scene = draft.scene.trim();
  if (scene) return `（开场）请根据开场场景直接开始扮演，不要用「好的」之类元话语。开场场景：${scene}`;
  if (isMultiDraft(draft)) return `（开场）请以在场所有角色的身份开始这一幕，描写当下环境与第一句对话。`;
  return `（开场）请以角色身份开始这一幕，描写当下环境与第一句对话。`;
}
